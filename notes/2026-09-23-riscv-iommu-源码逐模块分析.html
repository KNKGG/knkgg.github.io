                                                                                                                                        1.1 超时宏（:31–34）

                                                                                                                                        #define RISCV_IOMMU_QCSR_TIMEOUT       150000   // 150 ms
#define RISCV_IOMMU_QUEUE_TIMEOUT      150000   // 150 ms
#define RISCV_IOMMU_DDTP_TIMEOUT      10000000  // 10 s
#define RISCV_IOMMU_IOTINVAL_TIMEOUT  90000000  // 90 s

                                                                                                                                        四条超时对应四类硬件轮询场景，量级差异本身就有信息量：

                                                                                                                                        • QCSR / QUEUE（150 ms）：队列 CSR 位、队列指针寄存器，纯 MMIO 读回，硬件应立即响应。
                                                                                                                                        • DDTP（10 s）：设备目录表指针寄存器，切换 DDT 模式时硬件要重建目录缓存，慢得多。
                                                                                                                                        • IOTINVAL（90 s）：最慢，因为要等硬件彻底清空 IOTLB 缓存项，缓存项越多越久。后面 iodir_set_mode 和 iotlb_inval 里同步都走这条。

                                                                                                                                        #define RISCV_IOMMU_DEF_CQ_COUNT  8192
#define RISCV_IOMMU_DEF_FQ_COUNT  4096

                                                                                                                                        1.3 PPN ↔ 物理地址转换（:41–42）

                                                                                                                                        #define phys_to_ppn(pa)  (((pa) >> 2) & (((1ULL << 44) - 1) << 10))
#define ppn_to_phys(pn)  (((pn) << 2) & (((1ULL << 44) - 1) << 12))

                                                                                                                                        这是整份驱动最底层的位运算，注释 PHYS <=> PPN[53:10] 点明出处。PPN 在 RISC-V IOMMU 寄存器里的规范位是 [53:10]，隐含低 [9:0]=0（4 KiB 对齐）。逐位拆开：

                                                                                                                                        • phys_to_ppn(pa)：(pa >> 2) 等价于把物理地址的 bit 12..55 平移到 bit 10..53；掩码 ((2^44-1) << 10) 保留这 44 位，其余清零。净效果 = (pa >> 12) << 10，即「页号 × 2^10」填入 PPN 字段。
                                                                                                                                        • ppn_to_phys(pn)：逆过程，(pn << 2) 等价 (pn >> 10) << 12，把 PPN 字段还原成 4 KiB 对齐的物理地址。
                                                                                                                                        关键约束：44 位页号

                                                                                                                                        44 位页号 × 4 KiB = 2^56 的物理地址空间。RISC-V 物理地址上限 56 位（Sv57），所以这里封顶 44 位 PPN 正好匹配。

    atomic_t prod;              // 无界 producer 分配索引
    atomic_t head;              // 无界 shadow consumer 索引
    atomic_t tail;              // 无界 shadow producer 索引
    raw_spinlock_t lock;        // 序列化队列发布

    unsigned int irq;           // 分配的中断号

    void *base;                 // ring buffer 内核指针

                                                                                                                                        #define RISCV_IOMMU_QUEUE_INIT(q, name) do {                 \
    struct riscv_iommu_queue *_q = q;                        \
    _q->qid  = RISCV_IOMMU_INTR_ ## name;                    \
    _q->qbr  = RISCV_IOMMU_REG_ ## name ## B;                \

                                                                                                                                        2.2 四个关键偏移宏（:119–122）

                                                                                                                                        #define Q_HEAD(q)  ((q)->qbr + (RISCV_IOMMU_REG_CQH - RISCV_IOMMU_REG_CQB)) // head 寄存器偏移
#define Q_TAIL(q)  ((q)->qbr + (RISCV_IOMMU_REG_CQT - RISCV_IOMMU_REG_CQB)) // tail 寄存器偏移
#define Q_ITEM(q, index) ((q)->mask & (index))                             // 无界 → 环形下标
#define Q_IPSR(q)  BIT((q)->qid)                                           // 该队列在 IPSR 里的 pending 位
                                                                                                                                        head/tail 偏移对所有队列相同

                                                                                                                                        Q_HEAD/Q_TAIL 用 CQH-CQB / CQT-CQB 这个固定差值加到各自 qbr 上——三个队列（CQ/FQ/PQ）的基址寄存器到 head/tail 寄存器的间距在硬件里是统一设计的。

                                                                                                                                        2.3 无界 shadow 索引 —— 核心设计

                                                                                                                                        软件里维护的 head/tail/prod 永不按队列长度回绕，只有读写硬件寄存器、访问 ring buffer 时才 & mask 转成环形下标。这样「占用了多少条」就变成一句整数减法。

                                                                                                                                        硬件寄存器 / 内存下标软件 shadow 索引
                                                                                                                                        回绕周期mask+1（默认 8192 / 4096 条）2³²（约 42.9 亿次）
                                                                                                                                        谁在绕硬件 head/tail、& mask 后的下标atomic_t 这个 32 位变量本身
                                                                                                                                        判断有界索引（经典环形）无界 shadow（本驱动）
                                                                                                                                        占用条目数(tail - head + size) % sizetail - head
                                                                                                                                        是否有空位((tail + 1) & mask) != head(prod - head) < mask
                                                                                                                                        是否已完成需处理回绕(int)(cons - index) > 0
                                                                                                                                        正确性靠「距离 < 2³¹」，不是「永不溢出」

                                                                                                                                        atomic_t 是 32 位，索引撑到 2³² 终究会溢出回绕。但驱动所有关键判断都用有符号差值（如 (int)(tail - head)），利用模 2³² 补码算术——只要 tail 与 head 的真实距离 < 2³¹，即便两者本身溢出绕回，差值依然正确。这个不变量由队列容量上限保证（代码注释 should be <= INT_MAX，默认仅 8192/4096 条）。

                                                                                                                                        2.4 queue_alloc 四件事（:132–196）

                                                                                                                                        1. WARL 探测上限：写 LOG2SZ 全 1（0x1F），读回硬件合法化值，取 min(软件默认, 硬件上限)。
                                                                                                                                        2. 定 ring buffer 位置：读回的 base 若 PPN 非零 → 硬件固定 IO 地址，devm_ioremap 映射；否则软件 riscv_iommu_get_pages 分配连续内存。
                                                                                                                                        3. 回写并验证：phys_to_ppn(phys) | LOG2SZ 写回，读回比对，不等则 -ENODEV。
                                                                                                                                        4. 更新 mask：mask = (2U << logsz) - 1。

                                                                                                                                        2.5 硬中断 queue_ipsr（:199–207）与 queue_vec（:209–213）

                                                                                                                                        static irqreturn_t riscv_iommu_queue_ipsr(int irq, void *data)
{
    if (riscv_iommu_readl(queue->iommu, RISCV_IOMMU_REG_IPSR) & Q_IPSR(queue))
        return IRQ_WAKE_THREAD;   // 命中本队列 pending 位 → 唤醒线程
    return IRQ_NONE;
}

static int riscv_iommu_queue_vec(struct riscv_iommu_device *iommu, int n)
{
    return (iommu->icvec >> (n * 4)) & RISCV_IOMMU_ICVEC_CIV;  // 复用 CIV 掩码取向量
}

                                                                                                                                        硬中断只做一件事：读 IPSR 判断是不是本队列的中断，是就 IRQ_WAKE_THREAD 唤醒下半部线程，不是就 IRQ_NONE（共享中断场景）。queue_vec 从 icvec 里按 qid*4 位偏移取该队列对应的中断向量。

                                                                                                                                        2.8 消费侧：consume / release / cons（:312–364）

                                                                                                                                        static int riscv_iommu_queue_consume(struct riscv_iommu_queue *queue, unsigned int *index)
{
    head = atomic_read(&queue->head);  tail = atomic_read(&queue->tail);
    last = Q_ITEM(queue, tail);
    available = (int)(tail - head);                 // 先用 shadow 算
    *index = head;
    if (available > 0) return available;

    // shadow 里没了，才读硬件 tail（有界），有变化则更新 shadow tail
    riscv_iommu_readl_timeout(queue->iommu, Q_TAIL(queue), tail,
                              (tail & ~queue->mask) == 0, 0, QUEUE_TIMEOUT);
    if (tail == last) return 0;
    return (int)(atomic_add_return((tail - last) & queue->mask, &queue->tail) - head);
}
                                                                                                                                        static void riscv_iommu_queue_release(struct riscv_iommu_queue *queue, int count)
{
    const unsigned int head = atomic_add_return(count, &queue->head);
    riscv_iommu_writel(queue->iommu, Q_HEAD(queue), Q_ITEM(queue, head));  // 回写硬件 head
}

// 读硬件 head（有界），换算回无界 consumer
static unsigned int riscv_iommu_queue_cons(struct riscv_iommu_queue *queue)
{
    cons = atomic_read(&queue->head);
    last = Q_ITEM(queue, cons);                       // 无界 → 有界快照
    riscv_iommu_readl_timeout(queue->iommu, Q_HEAD(queue), head,
                              !(head & ~queue->mask), 0, QUEUE_TIMEOUT);
    return cons + ((head - last) & queue->mask);      // 有界差值 → 加回无界
}

                                                                                                                                        回绕处理被收敛到唯一躲不掉的地方——读硬件寄存器（硬件索引是有界的）。假设 last=6，硬件 head 绕回成 1：(1 - 6) & 7 = (-5) & 7 = 3，正确算出硬件又消费了 3 条。

                                                                                                                                        2.9 queue_wait 与 queue_wait_for_space（:367–419）

                                                                                                                                        static int riscv_iommu_queue_wait(struct riscv_iommu_queue *queue,
                                  unsigned int index, unsigned int timeout_us)
{
    cons = atomic_read(&queue->head);
    flags = CQCSR_CQMF | CQCSR_CMD_TO | CQCSR_CMD_ILL;
    if ((int)(cons - index) > 0) return 0;            // 已被消费
    return readx_poll_timeout(riscv_iommu_queue_cons, queue, cons,
            (readl(queue->iommu, queue->qcr) & flags) ||
            (int)(cons - index) > 0, 0, timeout_us);  // 轮询硬件 head 或 CSR 报错
}

                                                                                                                                        queue_wait 的退出条件二选一：消费者索引越过提交点，或 CSR 报错（CQMF/CMD_TO/CMD_ILL）。queue_wait_for_space 在队列满时轮询硬件 head 等待空位，并在 raw_spin_lock 保护下重读 hw_head 推进 shadow head——注释特别强调必须在锁内重读，否则用锁前的旧快照可能因回绕把 shadow head 推进到硬件还没消费的条目。

                                                                                                                                        2.10 queue_send 入队四步（:422–476）

                                                                                                                                        // ① 锁内预留槽位（满了 wait_for_space 自旋）
for (;;) {
    raw_spin_lock_irqsave(&queue->lock, flags);
    prod = atomic_read(&queue->tail);  head = atomic_read(&queue->head);
    if ((prod - head) < queue->mask) break;
    head = Q_ITEM(queue, head);
    raw_spin_unlock_irqrestore(&queue->lock, flags);
    if (riscv_iommu_queue_wait_for_space(queue, head)) goto err_busy;
}
// ② 写入 ring buffer（此刻 tail 未动，IOMMU 看不到）
memcpy(queue->base + Q_ITEM(queue, prod) * entry_size, entry, entry_size);
// ③ dma_wmb()：数据先于 tail 门铃可见
dma_wmb();
riscv_iommu_writel(queue->iommu, Q_TAIL(queue), Q_ITEM(queue, prod + 1));
// ④ mmiowb() 确保门铃落地，之后才更新 shadow
mmiowb();
atomic_set(&queue->tail, prod + 1);
atomic_set(&queue->prod, prod + 1);

                                                                                                                                        核心是「先写数据、再敲门铃」，中间两道屏障钉死顺序：dma_wmb()（fence w,w）保证 memcpy 写入先于 tail 门铃；mmiowb()（fence o,w）保证门铃写真正落地之后才更新内存 shadow 计数。

                                                                                                                                        3.0 命令格式（iommu-bits.h:459–462）

                                                                                                                                        一条命令 16 字节，两个 64 位字段。 dword0[6:0] 是 opcode（命令大类），dword0[9:7] 是 func（子功能）。规范定义 4 类 opcode：

                                                                                                                                        opcode命令作用func
                                                                                                                                        1IOTINVAL页表/IOTLB 缓存失效VMA / GVMA
                                                                                                                                        2IOFENCE命令队列栅栏C
                                                                                                                                        3IODIR目录缓存失效INVAL_DDT / INVAL_PDT
                                                                                                                                        4ATSPCIe ATSINVAL / PRGR

                                                                                                                                        本模块只涉及 IOFENCE.C，构造函数在 iommu-bits.h:766–772：

                                                                                                                                        static inline void riscv_iommu_cmd_iofence(struct riscv_iommu_command *cmd)
{
    cmd->dword0 = FIELD_PREP(CMD0_OPCODE, CMD_IOFENCE_OPCODE) |
                  FIELD_PREP(CMD0_FUNC, CMD_IOFENCE_FUNC_C) |
                  CMD0_IOFENCE_PR | CMD0_IOFENCE_PW;   // 关键：PR | PW
    cmd->dword1 = 0;
}

                                                                                                                                        PR（读栅栏）+ PW（写栅栏）同时置位，表示这条 fence 要等此前所有命令的读/写副作用都完成才算执行完——一个完整同步点。

                                                                                                                                        3.1 cmdq_process（:483–508）—— CQ 中断 thread

                                                                                                                                        这是命令队列注册的中断处理函数。核心要理解的是：这个中断不负责通知「命令完成」，只负责上报错误。

                                                                                                                                        四个错误位（iommu-bits.h:152–161）：CQMF（内存 fault）、CMD_TO（命令执行超时）、CMD_ILL（非法命令）、FENCE_W_IP（fence 写 pending）。

                                                                                                                                        设计取舍

                                                                                                                                        正常命令完成走软件轮询 head 寄存器（见 3.3 的 queue_wait），不走中断。中断只在错误发生时「推」一把，避免轮询周期漏掉错误位。所以这个 thread 里基本是错误处理 + 一个 /* Placeholder */，完整 RAS 恢复尚未实现。

                                                                                                                                        3.2 cmd_send（:511–515）

                                                                                                                                        riscv_iommu_queue_send(&iommu->cmdq, cmd, sizeof(*cmd), NULL);

                                                                                                                                        一行包装，把命令塞进 cmdq。fire-and-forget：不等待、不返回完成状态（out_prod=NULL）。所有需要「发完确认」的场景都用下面的 cmd_sync。

                                                                                                                                        3.3 cmd_sync（:518–536）—— 同步栅栏

                                                                                                                                        riscv_iommu_cmd_iofence(&cmd);                          // 构造 IOFENCE.C

    dev_err_once(...);

                                                                                                                                        语义链：命令队列是 FIFO 且硬件顺序执行 → 只要 IOFENCE.C 这条命令被硬件消费，就等价于它之前所有命令（如 IOTINVAL、IODIR.INVAL_DDT）都已执行完。这就是「失效 + 同步」两步里同步那一步的标准写法。

                                                                                                                                        queue_wait 的完成条件（:367–383）：(int)(cons - index) > 0（消费者已越过发送时的 prod），或 CQCSR 出现 CQMF/CMD_TO/CMD_ILL 错误。

                                                                                                                                        关键细节：cons 读的是硬件 head

                                                                                                                                        cons 来自 riscv_iommu_queue_cons（:352–364），它读的是硬件 head 寄存器，不是软件 shadow head。原因是 CQ 的消费者是硬件——硬件每执行完一条命令就推进 CQH；软件 shadow head 只服务于 FQ（软件自己消费 fault 记录）那一侧，两者方向相反。

                                                                                                                                        4.2 fltq_process（:556–588）—— FQ 中断 thread

                                                                                                                                        events = (struct riscv_iommu_fq_record *)queue->base;

riscv_iommu_writel(iommu, RISCV_IOMMU_REG_IPSR, Q_IPSR(queue));  // ① 清 FIP

do {
    cnt = riscv_iommu_queue_consume(queue, &idx);                 // ② 读硬件 tail，算可用条数
    for (len = 0; len < cnt; idx++, len++)
        riscv_iommu_fault(iommu, &events[Q_ITEM(queue, idx)]);    // ③ 逐条上报
    riscv_iommu_queue_release(queue, cnt);                        // ④ 推进硬件 head
} while (cnt > 0);                                               // ⑤ 循环直到清空

ctrl = riscv_iommu_readl(iommu, queue->qcr);                       // ⑥ 清 FQMF/FQOF 错误
if (ctrl & (FQCSR_FQMF | FQCSR_FQOF)) { ... dev_warn(...); }
                                                                                                                                        1. 方向反转：CQ 是软件→硬件（软件 producer、硬件 consumer）；FQ 是硬件→软件（硬件 producer 写 tail，软件 consumer 推进 head）。所以消费函数用的是 queue_consume（读硬件 tail）而不是 queue_cons（读硬件 head）。
                                                                                                                                        2. queue_consume 先看软件 shadow 里还有没有剩余（tail - head），没有才去读硬件 tail 寄存器，把硬件最新 producer 同步进 shadow tail，再算差值。
                                                                                                                                        3. do-while (cnt > 0) 循环：处理完一批后，处理期间硬件可能又写了新记录，所以要再 consume 一次，直到真的读空。这样一次中断把积压的 fault 全消化掉。
                                                                                                                                        4. queue_release 推进硬件 head 寄存器——告诉硬件「这些记录我读完了，ring 空间可回收」。硬件靠 head/tail 间距判断 FQ 是否满、是否会溢出（FQOF）。
                                                                                                                                        5. 溢出语义：FQOF（fault queue overflow）是 FQ 特有错误位（CQ 没有，CQ 是软件主导不会溢出）。一旦溢出，说明 fault 产生速度超过软件消费速度，有记录被丢弃。

                                                                                                                                        5.1 get_dc（:591–677）—— devid → DC 的查找/按需分配

                                                                                                                                        这是翻译入口的核心：给定 devid，返回它对应的 DC 指针，目录层级不存在就现场建。

                                                                                                                                        ② 层级范围校验（:628–630）

                                                                                                                                        • 迁移约束：只允许 BARE/OFF ↔ xLVL，不允许 xLVL 之间直切（中间必须先退到 BARE/OFF），避免目录结构在半切换状态下被硬件解析。

                                                                                                                                        1.                                                                                                                                         2. 逐地址 stride：9 * __ffs(combined) + 12。RISC-V 每级页表 9 位（512 项）、页大小 4 KiB（12 位）。__ffs(combined) 是最低被修改层级，stride 就是该层级一个条目覆盖的地址跨度。
                                                                                                                                        3. 512 条阈值：逐地址失效最多发 512 条命令，超过就退全局——发命令本身有开销，大范围逐条不如一条全局（AV=0）划算。

    WRITE_ONCE(dc->tc, tc);                              // ⑥ 最后写 TC.V 生效

                                                                                                                                        三个「为什么」：

                                                                                                                                        1. 为什么两遍 + 中间 sync：如果直接覆盖 fsc/ta，硬件可能正在用旧 DC 做翻译，新旧交错会产生瞬态错误翻译。所以先清 V 让旧上下文「退休」，cmd_sync 等硬件确认放弃，再写新值。
                                                                                                                                        2. 为什么 V 位最后写（⑤⑥）：TC.V 是 DC 的「生效开关」，dma_wmb() 保证 fsc/ta 的写先于 tc 到达硬件可见性，否则硬件会看到 V=1 但 fsc/ta 还是旧值的非法状态。
                                                                                                                                        3. 为什么每个 ID 失效两次：清 V 后失效一次（让硬件丢弃缓存的旧 DC），写新值后再失效一次（强制硬件重载新 DC）。

                                                                                                                                        11.4 全驱动闭环

                                                                                                                                        把 11 个模块串起来是一条完整数据流：

                                                                                                                                        启动: init → init_check(字节序/中断) → iodir_alloc(建DDT根)
      → queue_alloc×2 → queue_enable×2(CQ/FQ) → set_mode(切xLVL) → register ops
                          ↓
设备接入: probe_device(预建DC,不置V) → attach_paging_domain
          (bond_link → iodir_update: 清V→失效→写新→再失效)
                          ↓
翻译: 设备DMA → 硬件走DDT→DC→页表(第一级FSC/第二级IOHGATP)
                          ↓
页表变更: unmap → iotlb_sync(tlbi_calc决策粒度 → iotlb_inval发失效 → cmd_sync等fence)
                          ↓
故障上报: 硬件写FQ → fltq_process(consume/release) → fault打印
                          ↓
收尾: remove → unregister → OFF → queue_disable

                                                                                                                                        13.2 write_msi_msg（:24–44）—— 写内部 MSI 配置表

                                                                                                                                        平台设备 MSI 的核心：把一条 MSI 消息写进 IOMMU 内部的「MSI 配置表」。

                                                                                                                                        static void riscv_iommu_write_msi_msg(struct msi_desc *desc, struct msi_msg *msg)
{
    u16 idx = desc->msi_index;                      // 用向量索引当表项号
    u64 addr = ((u64)msg->address_hi << 32) | msg->address_lo;

    if (addr != (addr & RISCV_IOMMU_MSI_CFG_TBL_ADDR))   // [55:2]
        dev_err_once(...);                           // 地址被截断时告警（只报一次）

    addr &= RISCV_IOMMU_MSI_CFG_TBL_ADDR;            // 丢低 2 位

    riscv_iommu_writeq(iommu, RISCV_IOMMU_REG_MSI_CFG_TBL_ADDR(idx), addr);
    riscv_iommu_writel(iommu, RISCV_IOMMU_REG_MSI_CFG_TBL_DATA(idx), msg->data);
    riscv_iommu_writel(iommu, RISCV_IOMMU_REG_MSI_CFG_TBL_CTRL(idx), 0);
}
                                                                                                                                        为什么平台设备要自己写 MSI 消息？

                                                                                                                                        PCI 设备有标准的 MSI/MSI-X capability 结构，但平台设备没有。所以 RISC-V IOMMU 规范规定：当 IOMMU 作为平台设备要发 MSI 时，它内部有一张 MSI 配置表（RISCV_IOMMU_REG_MSI_CFG_TBL = 0x0300），软件把「地址 + 数据」写进去，硬件产生中断事件时就把表里的消息发到对应地址。这是两个接入层最本质的区别。

                                                                                                                                        表项布局（每项 0x10 字节，_n 为向量索引）：

                                                                                                                                        偏移寄存器位宽含义
                                                                                                                                        +0x00MSI_CFG_TBL_ADDR(n)64目标地址 [55:2]（GENMASK_ULL(55,2)）
                                                                                                                                        +0x08MSI_CFG_TBL_DATA(n)32MSI 数据（GENMASK_ULL(31,0)）
                                                                                                                                        +0x0CMSI_CFG_TBL_CTRL(n)32控制位（CTRL_M = BIT(0)，mask 位）
                                                                                                                                        • idx = desc->msi_index：把 Linux 分配的 MSI 向量索引直接当表项号，一一对应 iommu->irqs[idx]。
                                                                                                                                        • 地址截断到 [55:2]：MSI 地址按规范 4 字节对齐，低 2 位恒为 0；高位超出 55 位则告警。对应规范里 MSI 地址字段就是 [55:2]。
                                                                                                                                        • CTRL 写 0：表项默认未 mask（CTRL_M 位清零），让该中断立即可用。

                                                                                                                                        这个函数作为回调传给 platform_device_msi_init_and_alloc_irqs，在初始配置和 CPU 迁移（irqbalance 改 affinity）时都会被调用，把新消息重写到表里。
